RESPONSABLE IDENTIFIÉ PAR LE ROBOT DESSINATEUR

ATTENTION : UNE FOIS LE ROBOT LANCÉ NE TOUCHEZ SURTOUT PAS A LA SOURIS 

import java.awt.*;
import java.awt.event.*;
import java.nio.file.*;
import java.util.*;
import java.util.List;

public class DrawSegments {

    public static void main(String[] args) throws Exception {

        Dimension ecran = Toolkit.getDefaultToolkit().getScreenSize();

        new ProcessBuilder("mspaint").start();
        Thread.sleep(5000);

        Robot robot = new Robot();

        int decalageX = ecran.width/2 - 216;
        int decalageY = ecran.height/2 - 180;

        robot.mouseMove(decalageX, decalageY);
        robot.mousePress(InputEvent.BUTTON1_DOWN_MASK);
        robot.mouseRelease(InputEvent.BUTTON1_DOWN_MASK);

        List segments = chargerSegements("segments.txt");

        for (int[] s : segments) {
            robot.mouseMove(s[0] + decalageX, s[1] + decalageY);
            robot.mousePress(InputEvent.BUTTON1_DOWN_MASK);
            robot.mouseMove(s[2] + decalageX, s[3] + decalageY);
            robot.mouseRelease(InputEvent.BUTTON1_DOWN_MASK);
            Thread.sleep(35);
        }
    }

    static List chargerSegements(String fichier) throws Exception {

        List liste = new ArrayList<>();
        List lignes = Files.readAllLines(Paths.get(fichier));

        for (String line : lignes) {
            String[] p = line.split(",");

            int x1 = Integer.parseInt(p[0]);
            int y1 = Integer.parseInt(p[1]);
            int x2 = Integer.parseInt(p[2]);
            int y2 = Integer.parseInt(p[3]);

            liste.add(new int[]{x1, y1, x2, y2});
        }

        return liste;
    }
}
